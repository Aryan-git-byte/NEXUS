if(NEXUS_UNIVERSAL_XTENSA)
    # IDF 5.5 otherwise replaces a supplied CMAKE_TOOLCHAIN_FILE with its launcher.
    set(_NEXUS_UNIVERSAL_TOOLCHAIN "${CMAKE_CURRENT_LIST_DIR}/toolchains/universal/toolchain-esp32s3.cmake")
    macro(__target_set_toolchain)
        if(NOT IDF_TARGET STREQUAL "esp32s3")
            message(FATAL_ERROR "The NEXUS universal compiler option requires ESP32-S3")
        endif()
        set(CMAKE_TOOLCHAIN_FILE "${_NEXUS_UNIVERSAL_TOOLCHAIN}")
    endmacro()
endif()
